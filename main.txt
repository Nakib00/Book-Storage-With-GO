package main

import (
	"log"
	"net/http"

	"github.com/Nakib00/go-bookstore/pkg/routes"
	"github.com/gorilla/mux"
)

func main() {
	r := mux.NewRouter()
	routes.RegisterBookStoreRoutes(r)
	http.Handle("/", r)
	log.Printf("Starting server on port 9010\n")
	log.Fatal(http.ListenAndServe("localhost:9010", r))
}
